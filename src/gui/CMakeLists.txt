#
# NetShare GUI 模块
# 提供 Qt Quick/QML 用户界面
#

set(NETSHARE_GUI_MODULE_NAME NetshareGui)

# Qt QML policies
qt_policy(SET QTP0001 NEW)  # Resource prefix defaults to /qt/qml
qt_policy(SET QTP0003 NEW)  # RESOURCE_PREFIX defaults to module URI
qt_policy(SET QTP0004 NEW)  # qmldir auto-generated by qt_add_qml_module
qt_policy(SET QTP0005 NEW)  # qmllint auto-enabled

# =============================================================================
# C++ 源码
# =============================================================================
set(GUI_SOURCES
    QRCodeHelper.h
    QRCodeHelper.cpp
    ThemeProvider.h
    ThemeProvider.cpp
)

# =============================================================================
# 创建 Qt Quick 库
# =============================================================================
qt_add_library(${NETSHARE_GUI_MODULE_NAME} STATIC
    ${GUI_SOURCES}
)

# =============================================================================
# QML 模块注册 (替代 resources.qrc + qmldir)
# =============================================================================
qt_add_qml_module(${NETSHARE_GUI_MODULE_NAME}
    URI NetShare
    VERSION 1.0
    OUTPUT_DIRECTORY ${CMAKE_CURRENT_BINARY_DIR}/NetShare
    QML_FILES
        qml/Main.qml
        qml/ThemedButton.qml
        qml/ThemedComboBox.qml
        qml/ThemedSwitch.qml
        qml/ShareManagement.qml
        qml/ReceiveManagement.qml
        qml/TransferList.qml
        qml/DeviceDiscovery.qml
        qml/SettingsPage.qml
        qml/MessagePage.qml
    RESOURCES
        qml/icons/send.svg
        qml/icons/receive.svg
        qml/icons/transfer.svg
        qml/icons/lan.svg
        icons/netshare.png
        qml/themes/dark.json
        qml/themes/light.json
)

# =============================================================================
# 库配置
# =============================================================================
target_include_directories(${NETSHARE_GUI_MODULE_NAME} PUBLIC
    ${CMAKE_CURRENT_SOURCE_DIR}
)

target_include_directories(${NETSHARE_GUI_MODULE_NAME} PRIVATE
    ${CMAKE_SOURCE_DIR}/src
)

target_link_libraries(${NETSHARE_GUI_MODULE_NAME} PUBLIC
    Qt6::Core
    Qt6::Gui
    Qt6::Qml
    Qt6::Quick
    Qt6::QuickControls2
    Qt6::QuickDialogs2
    Qt6::QuickEffectsPrivate
    NetshareCore
    NetshareQrcode
)

# =============================================================================
# 导出
# =============================================================================
set_target_properties(${NETSHARE_GUI_MODULE_NAME} PROPERTIES
    OUTPUT_NAME "NetshareGui"
    POSITION_INDEPENDENT_CODE ON
)

message(STATUS "  GUI module: ${NETSHARE_GUI_MODULE_NAME} with QML module NetShare")
